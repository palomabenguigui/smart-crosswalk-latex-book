% Font + BiDi smoke test (P1). Compile from the repository root:
%   lualatex -output-directory=build scripts/font-test.tex
% Loads the bundled OFL Noto fonts from fonts/ by path (no system fonts).
\documentclass[11pt]{article}
\usepackage[a5paper,margin=15mm]{geometry}
\usepackage[english,bidi=basic]{babel}
\babelprovide[import,main]{hebrew}

\newcommand\fontopts[1]{Path=fonts/#1/, Extension=.otf}
% Noto Hebrew fonts contain ONLY Hebrew glyphs (no digits, Latin, punctuation).
% Fallback chains supply the missing characters from the matching Latin family.
\directlua{
  luaotfload.add_fallback("hebrm",   {"[fonts/NotoSerif/NotoSerif-Regular.otf]:mode=harf;"})
  luaotfload.add_fallback("hebrmbf", {"[fonts/NotoSerif/NotoSerif-Bold.otf]:mode=harf;"})
  luaotfload.add_fallback("hebsf",   {"[fonts/NotoSans/NotoSans-Regular.otf]:mode=harf;"})
  luaotfload.add_fallback("hebsfbf", {"[fonts/NotoSans/NotoSans-Bold.otf]:mode=harf;"})
}
% Latin
\babelfont{rm}[\fontopts{NotoSerif}, UprightFont=*-Regular, BoldFont=*-Bold,
  ItalicFont=*-Italic, BoldItalicFont=*-BoldItalic]{NotoSerif}
\babelfont{sf}[\fontopts{NotoSans}, UprightFont=*-Regular, BoldFont=*-Bold,
  ItalicFont=*-Italic, BoldItalicFont=*-BoldItalic]{NotoSans}
\babelfont{tt}[\fontopts{NotoSansMono}, UprightFont=*-Regular, BoldFont=*-Bold]{NotoSansMono}
% Hebrew
\babelfont[hebrew]{rm}[\fontopts{NotoSerifHebrew}, UprightFont=*-Regular, BoldFont=*-Bold,
  UprightFeatures={RawFeature={fallback=hebrm}}, BoldFeatures={RawFeature={fallback=hebrmbf}}]{NotoSerifHebrew}
\babelfont[hebrew]{sf}[\fontopts{NotoSansHebrew}, UprightFont=*-Regular, BoldFont=*-Bold,
  UprightFeatures={RawFeature={fallback=hebsf}}, BoldFeatures={RawFeature={fallback=hebsfbf}}]{NotoSansHebrew}
% Math letters (\mathrm) from the Latin serif, not from the Hebrew main font
\setmathrm{NotoSerif}[\fontopts{NotoSerif}, UprightFont=*-Regular, BoldFont=*-Bold, ItalicFont=*-Italic]
